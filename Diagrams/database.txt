@startuml
!theme plain
hide circle
skinparam linetype ortho

entity "users" {
  * id : serial
  --
  name : text
  email : text <<UNIQUE>>
  password_hash : text
}

entity "addresses" {
  * id : serial
  --
  street : text
  city : text
  neighborhood : text
  state : text
  cep : text
  number : integer
  complement : text DEFAULT 'nenhum'
}

entity "shops" {
  * id : serial
  --
  user_id : integer <<UNIQUE>> <<FK>>
  address_id : integer <<FK>>
  telephone : text
  slug : text
  name : text
  created_at : timestamp with time zone DEFAULT now()
  status : text DEFAULT 'active'
}

entity "consignees" {
  * id : serial
  --
  user_id : integer <<FK>>
  address_id : integer <<FK>>
  shop_id : integer <<FK>>
  telephone : text <<UNIQUE>>
}

entity "pickup_locations" {
  * id : serial
  --
  address_id : integer <<FK>>
  shop_id : integer <<FK>>
  name : text <<UNIQUE>>
  pickup_at : timestamp with time zone
}

entity "consignments" {
  * id : serial
  --
  shop_id : integer <<FK>>
  consignee_id : integer <<FK>>
  pickup_location_id : integer <<FK>>
  comission_type : text
  comission_value : numeric(12, 2)
  created_at : timestamp with time zone DEFAULT now()
  status : text
  code : uuid
  slug : text
}

entity "reports" {
  * id : serial
  --
  consignment_id : integer <<FK>>
  scope : text
  type : text
  priority : text
  description : text
  status : text
  created_at : timestamp with time zone DEFAULT now()
  consignee_id : integer
  moment : text
}

entity "categories" {
  * id : serial
  --
  name : text
  description : text
  shop_id : integer <<FK>>
  UNIQUE (shop_id, name)
}

entity "products" {
  * id : serial
  --
  name : text
  price : numeric(12, 2)
  discount : numeric(5, 2)
  stock : integer
  featured : boolean
  shop_id : integer <<FK>>
  moderation_status : text DEFAULT 'pending'
  category_id : integer <<FK>>
  description : text
  attr : jsonb DEFAULT '{}'
  code : text
  uniqueid : uuid DEFAULT uuidv4()
  platform_cut : numeric(12, 2)
  platform_cut_type : text
}

entity "packages" {
  * id : serial
  --
  product_id : integer <<FK>>
  consignment_id : integer <<FK>>
  quantity : integer
  price : numeric(12, 2)
  created_at : timestamp with time zone DEFAULT now()
  discount : numeric(12, 2)
  featured : boolean
  attributesvalues : json DEFAULT '{}'
}

entity "admins" {
  * id : serial
  --
  user_id : integer <<UNIQUE>> <<FK>>
}

entity "clients" {
  * id : serial
  --
  user_id : integer <<UNIQUE>> <<FK>>
}

entity "carts" {
  * id : serial
  --
  client_id : integer <<FK>>
  session_id : text <<UNIQUE>>
}

entity "cart_product" {
  * id : serial
  --
  product_id : integer <<FK>>
  cart_id : integer <<FK>>
  personalizations : jsonb
  quantity : integer
  price : numeric(12,2) DEFAULT 0
}

entity "payments" {
  * id : serial
  --
  cart_id : integer <<FK>>
  address_id : integer <<FK>>
  transaction_amount : numeric(12, 2)
  quantity : integer
  amount : numeric(12, 2)
  status : text
  description : text
  title : text
  payment_id : text
  payer_name : text
  payer_email : text
  payer_surname : text
  payment_method : text
  currency : text
  external_reference : text
  method : text
  metadata : jsonb
  processed_at : timestamp with time zone DEFAULT now()
}

entity "shipments" {
  * id : serial
  --
  address_id : integer <<FK>>
  payment_id : integer <<FK>>
  cart_id : integer <<FK>>
  shop_id : integer <<FK>>
  code : uuid <<UNIQUE>> DEFAULT uuidv4()
  status : text
  created_at : timestamp with time zone DEFAULT now()
}

entity "recovery_tokens" {
  * id : serial
  --
  user_id : integer <<FK>>
  token : text
  expiration_date : timestamp with time zone
  confirmation_code : text
  created_at : timestamp with time zone DEFAULT now()
}

' Relacionamentos
users ||--o| shops : "0..1 shop"
users ||--o{ consignees
users ||--o| admins : "0..1 admin"
users ||--o| clients : "0..1 client"
users ||--o{ recovery_tokens

addresses ||--o{ shops
addresses ||--o{ consignees
addresses |o--o{ pickup_locations
addresses |o--o{ payments
addresses |o--o{ shipments

shops ||--o{ consignees
shops ||--o{ pickup_locations
shops ||--o{ consignments
shops ||--o{ products
shops ||--o{ categories
shops ||--o{ shipments

consignees |o--o{ consignments
pickup_locations |o--o{ consignments

consignments ||--o{ reports
consignments ||--o{ packages

categories ||--o{ products

products ||--o{ packages
products ||--o{ cart_product

clients |o--o{ carts

carts ||--o{ cart_product
carts ||--o{ payments
carts ||--o{ shipments

payments ||--o{ shipments


@enduml
