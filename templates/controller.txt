package %(flat)

import (
	"linhazero.com/pvapi/models/%(flat)model"
	"linhazero.com/pvapi/util"
)

func ListHandler(
	pool *pgxpool.Pool,
) func(c *gin.Context) {
	return func(c *gin.Context) {
        templates, err := templatemodel.List(
            c.Request.Context(),
            pool,
        )

        if err != nil {
            util.HandleGuessFromDBOperation(c, err)
            return
        }

        collection := util.ResourceCollection(templates)

        // CHANGEME
        util.Wrap(
            c, "[template]s listados com sucesso",
            collection,
        )
	}
}

func CreateHandler(
	pool *pgxpool.Pool,
) func(c *gin.Context) {
	return func(c *gin.Context) {
        var request templatemodel.CreateTemplateRequest

        err := c.ShouldBind(&request)

        if err != nil {
            util.HandleBadRequest(c, err)
            return
        }

        template := request.Prepare()

        id, err := template.Save(c.Request.Context(), pool)

        if err != nil {
            util.HandleGuessFromDBOperation(c, err)
            return
        }

        // CHANGEME
        util.WrapLabeled(
            c, "[template] criado com sucesso",
            "id", id,
        )
	}
}

func ViewHandler(
	pool *pgxpool.Pool,
) func(c *gin.Context) {
	return func(c *gin.Context) {
        templateID := c.Param("template")

        template, err := templatemodel.Find(
            c.Request.Context(),
            pool,
            templateID,
        )

        if err != nil {
            util.HandleGuessFromDBOperation(c, err)
            return
        }

        // CHANGEME
        util.WrapLabeled(
            c, "[template] acessado com sucesso",
            "template": template.ToResource(),
        )
	}
}

func UpdateHandler(
	pool *pgxpool.Pool,
) func(c *gin.Context) {
	return func(c *gin.Context) {
        templateID := c.Param("template")

        var request templatemodel.UpdateTemplateRequest

        err := c.ShouldBind(&request)

        if err != nil {
            util.HandleBadRequest(c, err)
            return
        }

        template := request.Prepare(templateID)

        err := template.Save(c.Request.Context(), pool)

        if err != nil {
            util.HandleGuessFromDBOperation(c, err)
            return
        }

        // CHANGEME
        util.WrapLabeled(
            c, "[template] atualizado com sucesso",
            "template": template.ToResource(),
        )
	}
}

func DeleteHandler(
	pool *pgxpool.Pool,
) func(c *gin.Context) {
	return func(c *gin.Context) {
        err := templatemodel.Delete(
            c.Request.Context(), pool,
            c.Param("template"),
        )

        if err != nil {
            util.Handle500(c, err)
            return
        }

        util.OK(
            c, "[template] deletado com sucesso",
        )
	}
}
